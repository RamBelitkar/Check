class Item:
    """Represents an item with weight, value and ratio."""

    def __init__(self, index, weight, value):

        self.index = index                      # Item number

        self.weight = weight                    # Weight of item

        self.value = value                      # Profit/Value of item

        self.ratio = value / weight             # Profit per unit weight


def fractional_knapsack(capacity, items):
    """Fractional Knapsack using Greedy Algorithm"""

    items.sort(                                # Sort items by ratio
        key=lambda x: x.ratio,
        reverse=True
    )

    total_value = 0.0                          # Total profit

    current_weight = 0.0                       # Current weight in bag

    selection_details = []                     # Stores selected items

    for item in items:

        if capacity <= 0:                      # Stop when bag is full
            break

        if item.weight <= capacity:            # Item fully fits

            capacity -= item.weight            # Reduce remaining capacity

            current_weight += item.weight      # Add weight

            total_value += item.value          # Add full profit

            selection_details.append(
                (item.index,
                 item.weight,
                 item.value,
                 1.0)                          # 100% item taken
            )

        else:

            fraction = capacity / item.weight  # Fraction to take

            current_weight += item.weight * fraction

            total_value += item.value * fraction

            selection_details.append(
                (item.index,
                 item.weight,
                 item.value,
                 fraction)                     # Partial item taken
            )

            capacity = 0                       # Knapsack becomes full

    return total_value, selection_details      # Return answer


def main():

    print("=== Fractional Knapsack Problem ===")

    capacity = float(
        input("Enter Capacity: ")
    )

    n = int(
        input("Enter Number of Items: ")
    )

    items = []

    for i in range(1, n + 1):

        print(f"\nItem {i}")

        weight = float(
            input("Enter Weight: ")
        )

        value = float(
            input("Enter Value: ")
        )

        items.append(
            Item(i, weight, value)
        )

    max_value, details = fractional_knapsack(
        capacity,
        items
    )

    print("\nMaximum Profit =", round(max_value, 2))

    print("\nItems Selected:")

    for index, weight, value, fraction in details:

        print(
            f"Item {index} -> "
            f"{fraction*100:.1f}% Taken"
        )


if __name__ == "__main__":

    main()                                     # Start program

5. VIVA QUESTIONS & ANSWERS
Q1. What is Fractional Knapsack?

Answer: Fractional Knapsack allows taking an item completely or partially.

Q2. Which strategy is used?

Answer: Greedy Strategy.

Q3. Why Greedy works for Fractional Knapsack?

Answer: Selecting items with maximum profit per unit weight always leads to optimal solution.

Q4. What is the Greedy Choice?

Answer: Choose the item having highest Profit/Weight ratio first.

Q5. Formula for ratio?

Answer:

ratio = profit / weight

Q6. Which item is selected first?

Answer:

Item 5


because it has highest ratio:

8 / 1 = 8

Q7. Why is Item 4 partially selected?

Answer: Only 1 kg capacity remained, so only 20% of Item 4 was taken.

Q8. What is Time Complexity?

Answer:

O(n log n)


due to sorting.

Q9. What is Space Complexity?

Answer:

O(1)


or O(n) if storing selection details.

6. CONCLUSION
The Fractional Knapsack Problem is solved using
a Greedy Algorithm. Items are sorted according
to Profit/Weight ratio and selected in descending
order. If an item cannot fit completely, a fraction
of it is taken. For the given example, the maximum
profit obtained is 45.4 and the time complexity is
O(n log n).
