# =====================================================================
# 0-1 KNAPSACK PROBLEM USING DYNAMIC PROGRAMMING
# =====================================================================

def knapsack_01(capacity, weights, values, n):      # Function to solve 0-1 Knapsack

    dp = [[0 for _ in range(capacity + 1)]          # Create DP table
           for _ in range(n + 1)]                  # Size = (n+1) x (capacity+1)

    for i in range(1, n + 1):                      # Loop through all items

        for w in range(1, capacity + 1):           # Loop through capacities

            if weights[i - 1] <= w:               # Check if current item fits

                include_item = (                  # Value if item is included
                    values[i - 1] +
                    dp[i - 1][w - weights[i - 1]]
                )

                exclude_item = dp[i - 1][w]       # Value if item is excluded

                dp[i][w] = max(                   # Store maximum value
                    include_item,
                    exclude_item
                )

            else:

                dp[i][w] = dp[i - 1][w]           # Copy previous value

    max_value = dp[n][capacity]                   # Final maximum value

    w = capacity                                  # Start from last column

    selected_items = []                           # Store selected items

    for i in range(n, 0, -1):                     # Backtracking

        if max_value <= 0:                        # Stop if value becomes 0
            break

        if max_value == dp[i - 1]# Item not selected
            continue

        else:

            selected_items.append(i)              # Item selected

            max_value -= values[i - 1]            # Reduce obtained value

            w -= weights[i - 1]                   # Reduce remaining capacity

    return dp[n][capacity], selected_items        # Return answer


# =====================================================================
# MAIN FUNCTION
# =====================================================================

def main():

    print("=== 0-1 KNAPSACK PROBLEM ===")         # Title

    capacity = int(input("Enter Capacity: "))     # Input capacity

    n = int(input("Enter Number of Items: "))     # Input total items

    weights = []                                  # Weight list

    values = []                                   # Value list

    for i in range(n):                            # Input item details

        w = int(input(f"Weight of Item {i+1}: ")) # Read weight

        v = int(input(f"Value of Item {i+1}: "))  # Read value

        weights.append(w)                         # Store weight

        values.append(v)                          # Store value

    max_val, selected = knapsack_01(              # Call function
        capacity,
        weights,
        values,
        n
    )

    print("\nMaximum Value =", max_val)           # Display maximum value

    print("Selected Items =", selected)           # Display selected items


# =====================================================================
# DRIVER CODE
# =====================================================================

if __name__ == "__main__":                        # Program starts here

    main()

PRACTICAL / VIVA QUESTIONS AND ANSWERS
Q1. What is Knapsack Problem?

Answer: Knapsack Problem is an optimization problem where we maximise value without exceeding the given capacity.

Q2. What is 0-1 Knapsack?

Answer: Each item can either be selected completely (1) or not selected (0).

Q3. Which algorithm is used?

Answer: Dynamic Programming.

Q4. Why Dynamic Programming?

Answer: Because the problem contains:

Optimal Substructure
Overlapping Subproblems
Q5. What does dp[i][w] represent?

Answer: Maximum value obtained using first i items and capacity w.

Q6. What is the recurrence relation?

Answer:

dp[i][w] = max(
value[i] + dp[i-1][w-weight[i]],
dp[i-1][w]
)

Q7. What is Time Complexity?

Answer:

O(n × W)


where:

n = Number of Items
W = Capacity

Q8. What is Space Complexity?

Answer:

O(n × W)

Q9. Why is it called 0-1 Knapsack?

Answer: Because items cannot be divided. They are either selected or not selected.

Q10. What is Optimal Substructure?

Answer: Optimal solution can be built from optimal solutions of smaller subproblems.

Q11. What are Overlapping Subproblems?

Answer: Same subproblems are solved multiple times.

Q12. Why Greedy Algorithm is not used?

Answer: Greedy does not always provide the optimal solution for 0-1 Knapsack.

Q13. What is Backtracking?

Answer: Process of finding which items are selected from DP table.

Q14. Where is final answer stored?

Answer:

dp[n][capacity]

Q15. Can an item be selected multiple times?

Answer: No.

Q16. What are real-life applications?

Answer:

Cargo Loading
Memory Allocation
Resource Management
Investment Planning
Budget Optimisation
Q17. Is 0-1 Knapsack a Greedy Algorithm?

Answer: No, it is generally solved using Dynamic Programming.

Q18. What happens when item's weight exceeds capacity?

Answer: The item is not selected.

Q19. What is DP Table?

Answer: A table that stores solutions of smaller subproblems.

Q20. What is the objective of Knapsack Problem?

Answer: To maximise total value without exceeding capacity.

CONCLUSION

The 0-1 Knapsack Problem is solved using Dynamic Programming. A DP table is constructed to store intermediate solutions. For every item, we decide whether to include or exclude it. The final solution gives the maximum obtainable value without exceeding the given capacity. The time complexity is O(n × W) and the space complexity is O(n × W).
